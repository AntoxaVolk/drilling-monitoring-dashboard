Схема базы данных (PostgreSQL)
-- Таблица скважин CREATE TABLE wells ( well_id SERIAL PRIMARY KEY, well_name VARCHAR(50), field_name VARCHAR(100), cluster_no INT, well_type VARCHAR(20), spud_date DATE, td_planned FLOAT -- проектная глубина, м ); -- Инклинометрия CREATE TABLE inclinometry ( id SERIAL PRIMARY KEY, well_id INT REFERENCES wells(well_id), md FLOAT, -- измеренная глубина inclination FLOAT, -- зенитный угол, ° azimuth FLOAT, -- азимут, ° tvd FLOAT, -- вертикальная глубина northing FLOAT, easting FLOAT, dls FLOAT, -- интенсивность набора °/30м ts_utc TIMESTAMPTZ DEFAULT NOW() ); -- Параметры бурения (1 запись/мин) CREATE TABLE drilling_params ( id SERIAL PRIMARY KEY, well_id INT REFERENCES wells(well_id), ts_utc TIMESTAMPTZ, md FLOAT, rop FLOAT, -- механическая скорость wob FLOAT, -- нагрузка на долото, тс rpm INT, -- обороты torque FLOAT, -- момент, кН·м flow_rate FLOAT, -- расход, л/с spp FLOAT -- давление СПП, бар ); -- LWD-кривые CREATE TABLE lwd_curves ( id SERIAL PRIMARY KEY, well_id INT REFERENCES wells(well_id), md FLOAT, gr FLOAT, -- гамма-каротаж, API res_deep FLOAT, -- УЭС глубинное res_med FLOAT, -- УЭС среднее res_sh FLOAT -- УЭС мелкое );
Аналитические SQL-запросы
-- Среднее РОП по интервалам бурения SELECT FLOOR(md / 100) * 100 AS md_interval, ROUND(AVG(rop)::numeric, 2) AS avg_rop, ROUND(AVG(wob)::numeric, 2) AS avg_wob, COUNT(*) AS records FROM drilling_params WHERE well_id = 247 AND rop > 0 GROUP BY md_interval ORDER BY md_interval; -- Выявление аномалий (ГНВП-риск) SELECT ts_utc, md, spp, LAG(spp) OVER (ORDER BY ts_utc) AS prev_spp, spp - LAG(spp) OVER (ORDER BY ts_utc) AS delta_spp FROM drilling_params WHERE well_id = 247 AND spp - LAG(spp) OVER ( ORDER BY ts_utc) > 20 ORDER BY ts_utc DESC; -- NPT (НПВ) анализ за период SELECT DATE_TRUNC('day', ts_utc) AS day, SUM(CASE WHEN rop = 0 THEN 1 ELSE 0 END) AS npt_minutes, ROUND(100.0 * SUM(CASE WHEN rop = 0 THEN 1 ELSE 0 END) / COUNT(*), 1) AS npt_pct FROM drilling_params WHERE well_id = 247 GROUP BY day ORDER BY day; -- Литология по ГК (коллектор vs глина) SELECT md, CASE WHEN gr < 45 THEN 'Нефтеносный песчаник' WHEN gr < 75 THEN 'Алевролит' ELSE 'Аргиллит/глина' END AS lithology FROM lwd_curves WHERE well_id = 247 ORDER BY md;
